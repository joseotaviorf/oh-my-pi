# Version pins that MUST match the ETL runtime.
#
# The point of this image is that a notebook and the Airflow DAG it mirrors
# produce the same answer. Anything that changes numeric or serialization
# behaviour is pinned here rather than allowed to float to "newest".
#
# Sources:
#   packages/bietlejuice-runtime/envs/dbr-16-4/pyproject.toml  (DBR 16.4 LTS pins)
#   packages/bietlejuice-compiler/scripts/emr_init_script.sh   (EMR 7.12 bootstrap)
#   dags/**/*_cluster.yml                                      (per-DAG pins)
#
# Everything NOT listed here is free to resolve to the newest release that is
# compatible with these constraints on Python 3.9.

# numpy 2.x changes dtype promotion and repr; DBR 16.4 and 3 DAGs pin 1.26.4.
# Floating this to 2.x would silently diverge notebook results from ETL results.
numpy==1.26.4

# Arrow format compatibility with the bundled PySpark; DBR 16.4 pins 15.0.2.
pyarrow==15.0.2

# emr_init_script.sh: runtime wheel omits psycopg2 (DBR bundles it), EMR does not.
psycopg2-binary==2.9.9

# urllib3 2.x breaks the awscli/botocore shipped in the EMR base image.
urllib3>=1.25.4,<1.27

# Same reason — emr_init_script.sh restores this pin after the big resolve.
python-dateutil>=2.1,<=2.9.0

# bietlejuice-core needs >=2.32.3; validations-engine declares 2.28.1 and is
# installed --no-deps to avoid the downgrade (mirrors uv override-dependencies).
requests>=2.32.3
